% hyperref is already loaded above, with options; loading it again would be a
% no-op that silently discards them, so only configure it here.
%
% These read the Pandoc variables directly rather than \thetitle/\theauthor.
% Those two come from the `titling` package, which is not loaded here, so they
% expanded to nothing without raising an error and every PDF this template
% produced went out with no Title and no Author in its metadata. PDF/UA wants
% a document title, and it is what a screen reader announces when the file
% opens.
%
% pdflang sets the PDF catalog's /Lang, which decides how that text is
% pronounced.
% pdfdisplaydoctitle makes a reader show and announce the document's title
% rather than its filename. PDF/UA requires it, and without it a screen reader
% opens this file as "syllabus.pdf".
\hypersetup{
    pdftitle={$title$$if(subtitle)$ — $subtitle$$endif$},
    pdfauthor={$author$},
    pdfdisplaydoctitle=true,
$if(lang)$
    pdflang={$lang$},
$endif$
}

% Images. Pandoc wraps every figure in \pandocbounded, which its stock template
% defines and a custom one must; tools/tikz.lua emits it too. Without this any
% handout with a figure fails with "Undefined control sequence". Definition
% copied from Pandoc's default LaTeX template.
\usepackage{graphicx}
\makeatletter
\newsavebox\pandoc@box
\newcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi}
\makeatother

% Pandoc emits \tightlist for lists with no blank lines between items. Its
